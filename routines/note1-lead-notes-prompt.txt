```
def ig_handle(raw):
    s = (raw or "").strip()
    if not s: return None
    if "instagram.com" in s.lower():
        s = re.sub(r"(?i)^.*instagram\.com/(_u/|accounts/)?", "", s).split("?")[0].split("/")[0]
    else:
        s = re.sub(r"^https?://", "", s)
        s = re.sub(r"(?i)^(insta(gram)?|ig|id|handle|user(name)?)\s*[:\-=]?\s*", "", s)   # "Insta: xyz", "IG - xyz", "ID xyz"
        if re.match(r"^[^@\s]+@[^@\s]+\.[A-Za-z]{2,}$", s):
            s = s.split("@")[0]      # an email -> the part before @
        elif "@" in s:
            s = s.split("@")[-1]     # "@handle" / "name @handle" -> after the last @
    s = re.sub(r"[^A-Za-z0-9._]", "", s)      # drops spaces: "fit with priya" -> "fitwithpriya"
    s = s.strip(".").lower()
    return s or None
```
